#!/bin/bash

# Set the desired image name and size
IMAGE_NAME="MyNet_SysPrint.img"
IMAGE_SIZE="1TB"

# Print a message to the console
echo "Creating blank image..."

# Create the blank image file
dd if=/dev/zero of="$IMAGE_NAME" bs=1M count=$(echo "$IMAGE_SIZE" | sed 's/[^0-9]*//')

# Print a success message
echo "Image created successfully!"

# 2. Partition the image (using parted)
echo "Partitioning the image..."
sudo parted -s "$IMAGE_NAME" mklabel msdos
sudo parted -s "$IMAGE_NAME" mkpart primary "$FILE_SYSTEM" 1MiB 100%

# 3. Format the partition with the specified file system
echo "Formatting the partition..."
sudo kpartx -a -v "$IMAGE_NAME" # Map the partitions
PARTITION_DEVICE=$(sudo kpartx -l "$IMAGE_NAME" | awk '{print "/dev/mapper/"$1"p1"}') # Get the partition device
sudo mkfs."$FILE_SYSTEM" "$PARTITION_DEVICE"

# 4. Mount the partition to a mount point
echo "Mounting the partition..."
MOUNT_POINT="./my_image_mount"
mkdir -p "$MOUNT_POINT"
sudo mount "$PARTITION_DEVICE" "$MOUNT_POINT"

# 5. Use the mounted file system (example: create a file)
echo "Creating a test file..."
sudo touch "$MOUNT_POINT/testfile.txt"

# 6. Unmount and cleanup
echo "Unmounting and cleaning up..."
sudo umount "$MOUNT_POINT"
sudo kpartx -d "$IMAGE_NAME"
echo "Done! The file system is created in $IMAGE_NAME and mounted to $MOUNT_POINT."
