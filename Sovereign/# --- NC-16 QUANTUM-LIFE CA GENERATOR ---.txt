import time
import json

import asyncio

# Grid Settings for the Thomson, GA "Lattice"
WIDTH, HEIGHT = 40, 40
LAT_BASE, LON_BASE = 33.4700, -82.5020
STEP = 0.0005 

def evolve(grid):
    new_grid = grid.copy()
    for y in range(HEIGHT):
        for x in range(WIDTH):
            # Neighborhood check (Toroidal/Wrapped)
            n_alive = np.sum(grid[(y-1):(y+2), (x-1):(x+2)]) - grid[y, x]
            if grid[y, x] == 1:
                if n_alive < 2 or n_alive > 3:
                    new_grid[y, x] = 0
            else:
                if n_alive == 3:
                    new_grid[y, x] = 1
    return new_grid

async def stream_ca():
    uri = "ws://localhost:8890"
    async with websockets.connect(uri) as ws:
        # Initialize with a random "Quantum Soup"
        grid = np.random.choice([0, 1], WIDTH*HEIGHT, p=[0.8, 0.2]).reshape(HEIGHT, WIDTH)
        
        while True:
            grid = evolve(grid)
            # Find all "Live" cells to paint
            live_cells = np.argwhere(grid == 1)
            
            for cell in live_cells:
                # Transduce Grid (y,x) to GPS (Lat, Lon)
                payload = {
                    "lat": LAT_BASE + (cell[0] * STEP),
                    "lon": LON_BASE + (cell[1] * STEP),
                    "type": "CA_Pulse"
                }
                await ws.send(json.dumps(payload))
            
            print(f"[*] Generation Complete: {len(live_cells)} cells entangled.")
            await asyncio.sleep(0.5) # Evolution Speed

if __name__ == "__main__":
    asyncio.run(stream_ca())
