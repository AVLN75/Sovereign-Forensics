function start_usbip() { echo "[+] Loading kernel modules..." sudo modprobe usbip_core sudo modprobe usbip_host echo "[+] Starting usbip daemon..." # -D runs it in the background sudo usbipd -D echo "[+] Binding device $BUSID to network..." sudo usbip bind -b "$BUSID" echo "[!] Success: Device $BUSID is now available on the network." echo "[!] Port: 3240 (Default)" } function stop_usbip() { echo "[-] Unbinding device $BUSID..." sudo usbip unbind -b "$BUSID" echo "[-] Killing daemon and unloading modules..." sudo pkill usbipd sudo modprobe -r usbip_host sudo modprobe -r usbip_core echo "[+] System cleaned." } case "$1" in start) start_usbip ;; stop) stop_usbip ;; *) echo "Usage: $0 {start|stop}"; exit 1 ;; esac
