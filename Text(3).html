Partitions

Get-Partition | Where-Object { $_.DriveLetter -eq 'D' }

Partition 1: Storage (1TB)

New-Partition -DiskNumber 2 -Size 1TB -DriveLetter S | Format-Volume -FileSystem NTFS -NewFileSystemLabel "Storage"

Partition 2: VMs (500GB)

New-Partition -DiskNumber 0 -Size 500GB -DriveLetter V | Format-Volume -FileSystem NTFS -NewFileSystemLabel "VMs"

Partition 3: VM Swap Space (100GB)

New-Partition -DiskNumber 0 -Size 100GB -DriveLetter W | Format-Volume -FileSystem NTFS -NewFileSystemLabel "VM Swap"
